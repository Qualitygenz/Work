local CoreGui = game:GetService("CoreGui")
local HttpService = game:GetService("HttpService")
local Players = game:GetService("Players")
local LocalPlayer = Players.LocalPlayer

local ConfigFileName = "PigHub_KeyData.json"
local MainScriptURL = "https://api.mclo.gs/1/raw/PLc5Q7t"

local function fetchKeysFromURL()
    local success, data = pcall(function()
        return game:HttpGet("https://api.mclo.gs/1/raw/jA0SMTi", true)
    end)
    
    if not success or not data then
        return nil
    end
    
    local func = loadstring(data)
    if not func then
        return nil
    end
    
    func()
    return _G.PIGHUB_KEYS
end

local function saveKey(key, keyType)
    local data = {
        key = key,
        keyType = keyType,
        timestamp = os.time()
    }
    pcall(function()
        writefile(ConfigFileName, HttpService:JSONEncode(data))
    end)
end

local function loadSavedKey()
    local success, content = pcall(readfile, ConfigFileName)
    if not success or not content then return nil end
    
    local success2, data = pcall(HttpService.JSONDecode, HttpService, content)
    if not success2 or not data then return nil end
    
    return data
end

local function clearSavedKey()
    pcall(function()
        delfile(ConfigFileName)
    end)
end

local function validateKey(key, keysTable)
    if not keysTable or not key then return false, nil end
    
    local keyType = keysTable[key]
    if keyType then
        return true, keyType
    end
    return false, nil
end

local function loadMainScript()
    local success, scriptContent = pcall(function()
        return game:HttpGet(MainScriptURL, true)
    end)
    
    if not success or not scriptContent then
        warn("[KEY SYSTEM] ไม่สามารถโหลดสคริปต์หลักได้")
        return false
    end
    
    local func = loadstring(scriptContent)
    if not func then
        warn("[KEY SYSTEM] สคริปต์หลักมีข้อผิดพลาด")
        return false
    end
    
    func()
    return true
end

local function showKeyStatus(keyType)
    local screenGui = Instance.new("ScreenGui")
    screenGui.Name = "PigHubKeyStatus"
    screenGui.ResetOnSpawn = false
    screenGui.Parent = CoreGui
    
    local frame = Instance.new("Frame")
    frame.Size = UDim2.new(0, 180, 0, 25)
    frame.Position = UDim2.new(0, 10, 0, 10)
    frame.BackgroundColor3 = Color3.fromRGB(10, 8, 20)
    frame.BackgroundTransparency = 0.2
    frame.Parent = screenGui
    
    local frameCorner = Instance.new("UICorner")
    frameCorner.CornerRadius = UDim.new(0, 8)
    frameCorner.Parent = frame
    
    local frameStroke = Instance.new("UIStroke")
    frameStroke.Thickness = 1
    frameStroke.Parent = frame
    
    local label = Instance.new("TextLabel")
    label.Size = UDim2.new(1, -10, 1, 0)
    label.Position = UDim2.new(0, 5, 0, 0)
    label.BackgroundTransparency = 1
    label.Text = "PIG HUB | " .. keyType
    label.TextColor3 = Color3.fromRGB(100, 255, 100)
    label.TextScaled = true
    label.Font = Enum.Font.GothamBold
    label.Parent = frame
    
    task.spawn(function()
        local t = 0
        while screenGui and screenGui.Parent do
            t = t + 0.02
            frameStroke.Color = Color3.fromHSV(t % 1, 0.8, 1)
            task.wait(0.05)
        end
    end)
    
    return screenGui
end

local function initKeySystem()
    local saved = loadSavedKey()
    
    if saved and saved.key then
        local keysTable = fetchKeysFromURL()
        local isValid, keyType = validateKey(saved.key, keysTable)
        
        if isValid then
            print("[KEY SYSTEM] โหลดคีย์สำเร็จ: " .. keyType)
            showKeyStatus(keyType)
            loadMainScript()
            return true
        else
            clearSavedKey()
        end
    end
    
    showKeyInputGUI(function(success, keyType)
        if success then
            print("[KEY SYSTEM] ใส่คีย์สำเร็จ: " .. keyType)
            showKeyStatus(keyType)
            loadMainScript()
        else
            warn("[KEY SYSTEM] ใส่คีย์ไม่สำเร็จ")
        end
    end)
    
    return false
end

initKeySystem()
